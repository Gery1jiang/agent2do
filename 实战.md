# 阶段三：项目实战指南

## 一、基础概念

### 3.1 从学习到实战的跨越

实战项目的核心挑战不是模型能力，而是**系统工程**：

- **错误处理**：工具失败、模型幻觉、超时如何降级
- **状态管理**：多轮对话、中断恢复
- **成本控制**：token 用量、缓存、批处理
- **用户体验**：流式输出、进度反馈

### 3.2 四个实战项目概览

|项目|核心技术|难度|预计时间|
|---|---|---|---|
|代码助手|ReAct + 代码执行沙箱|⭐⭐|1-2 天|
|知识库问答|RAG pipeline + 检索优化|⭐⭐⭐|2-3 天|
|浏览器 Agent|Computer Use / Playwright|⭐⭐⭐⭐|3-5 天|
|研究助手|多 agent 分工 + 报告生成|⭐⭐⭐⭐|3-5 天|

### 3.3 代码执行安全原则

代码助手必须在**沙箱**中执行用户代码：

- 禁止网络访问（或白名单）
- 限制文件系统权限
- 设置执行超时（通常 30s）
- 捕获所有异常，不暴露系统信息

### 3.4 RAG Pipeline 核心步骤

```
文档 → 分块（Chunking）→ 嵌入（Embedding）→ 存储
查询 → 嵌入 → 检索 TopK → 重排（Rerank）→ 生成
```

关键参数：chunk_size（512-1024），chunk_overlap（50-100），top_k（3-5）。

### 3.5 多 Agent 分工模式

- **流水线（Pipeline）**：A 输出作为 B 输入，串行
- **并行（Parallel）**：多个 agent 同时处理不同子任务
- **监督（Supervisor）**：主 agent 分配任务给子 agent，汇总结果
- **辩论（Debate）**：多 agent 对同一问题给出不同观点，主 agent 综合

```python
import sys
import io
import traceback
from contextlib import redirect_stdout, redirect_stderr

def safe_execute(code: str, timeout: int = 30) -> dict:
    """
    在受限环境中执行 Python 代码
    返回：{"stdout": str, "stderr": str, "error": str | None}
    """
    stdout_capture = io.StringIO()
    stderr_capture = io.StringIO()

    # 受限全局变量：只允许安全的内置函数
    safe_globals = {
        "__builtins__": {
            "print": print, "len": len, "range": range,
            "enumerate": enumerate, "zip": zip, "map": map,
            "filter": filter, "sorted": sorted, "sum": sum,
            "min": min, "max": max, "abs": abs, "round": round,
            "int": int, "float": float, "str": str, "bool": bool,
            "list": list, "dict": dict, "set": set, "tuple": tuple,
            "isinstance": isinstance, "type": type,
        }
    }

    error = None
    try:
        with redirect_stdout(stdout_capture), redirect_stderr(stderr_capture):
            exec(code, safe_globals)
    except Exception:
        error = traceback.format_exc()

    return {
        "stdout": stdout_capture.getvalue(),
        "stderr": stderr_capture.getvalue(),
        "error": error,
    }
```

**文件：`01_code_assistant/agent.py`**

```python
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode
from typing import TypedDict, Annotated
import operator
from sandbox import safe_execute

# --- 工具定义 ---
@tool
def execute_python(code: str) -> str:
    """
    执行 Python 代码并返回输出结果。
    适合：验证算法、测试函数、数学计算。
    代码应包含 print 语句输出结果。
    """
    result = safe_execute(code)
    if result["error"]:
        return f"执行错误:\n{result['error']}"
    output = result["stdout"] or "(无输出)"
    if result["stderr"]:
        output += f"\n警告: {result['stderr']}"
    return output

@tool
def write_file(filename: str, content: str) -> str:
    """将代码或内容写入文件"""
    try:
        with open(f"./output/{filename}", "w", encoding="utf-8") as f:
            f.write(content)
        return f"文件已写入：./output/{filename}"
    except Exception as e:
        return f"写入失败: {e}"

import os
os.makedirs("./output", exist_ok=True)

# --- Agent 构建 ---
tools = [execute_python, write_file]
llm = ChatOpenAI(model="gpt-4o", temperature=0.1).bind_tools(tools)

SYSTEM = """你是一个代码助手。工作方式：
1. 理解用户的编程需求
2. 编写代码解决问题
3. 用 execute_python 工具测试代码
4. 如果测试失败，分析错误并修复
5. 确认正确后，向用户展示最终代码和解释

始终先测试代码再交付。"""

class State(TypedDict):
    messages: Annotated[list, operator.add]

def agent_node(state: State):
    from langchain_core.messages import SystemMessage
    messages = [SystemMessage(content=SYSTEM)] + state["messages"]
    return {"messages": [llm.invoke(messages)]}

def router(state: State):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END

graph = StateGraph(State)
graph.add_node("agent", agent_node)
graph.add_node("tools", ToolNode(tools))
graph.set_entry_point("agent")
graph.add_conditional_edges("agent", router)
graph.add_edge("tools", "agent")
app = graph.compile()

# --- 运行示例 ---
def code_assistant(request: str):
    print(f"\n用户: {request}\n{'='*50}")
    result = app.invoke({"messages": [HumanMessage(content=request)]})
    for msg in result["messages"]:
        if hasattr(msg, "content") and msg.content and not hasattr(msg, "tool_calls"):
            print(f"\n助手:\n{msg.content}")

code_assistant("实现一个 LRU 缓存，容量为 3，并演示 get/put 操作")
```

### 项目 2：知识库问答（RAG Pipeline）

**文件：`02_knowledge_qa/ingest.py`**

```python
import os
from langchain_community.document_loaders import PyPDFLoader, DirectoryLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores import Chroma

def ingest_documents(docs_dir: str = "./docs", db_dir: str = "./chroma_db"):
    """将目录中的 PDF 文档入库"""

    # 1. 加载文档
    loader = DirectoryLoader(docs_dir, glob="**/*.pdf", loader_cls=PyPDFLoader)
    raw_docs = loader.load()
    print(f"加载了 {len(raw_docs)} 个文档页面")

    # 2. 分块：chunk_size 和 overlap 影响检索质量
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=800,
        chunk_overlap=100,
        separators=["\n\n", "\n", "。", ".", " ", ""],
    )
    chunks = splitter.split_documents(raw_docs)
    print(f"分割为 {len(chunks)} 个 chunks")

    # 3. 嵌入 + 存储
    embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    db = Chroma.from_documents(chunks, embeddings, persist_directory=db_dir)
    print(f"向量数据库已保存到 {db_dir}")
    return db

if __name__ == "__main__":
    os.makedirs("./docs", exist_ok=True)
    # 在 ./docs 目录放入你的 PDF 文件后运行
    ingest_documents()
```

**文件：`02_knowledge_qa/agent.py`**

```python
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, SystemMessage
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode
from typing import TypedDict, Annotated
import operator

# 加载向量库
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
db = Chroma(persist_directory="./chroma_db", embedding_function=embeddings)
retriever = db.as_retriever(search_kwargs={"k": 4})

@tool
def search_knowledge_base(query: str) -> str:
    """
    从知识库中检索与查询最相关的内容。
    当需要回答特定文档相关的问题时使用。
    """
    docs = retriever.invoke(query)
    if not docs:
        return "知识库中未找到相关内容。"
    results = []
    for i, doc in enumerate(docs, 1):
        source = doc.metadata.get("source", "未知来源")
        page = doc.metadata.get("page", "?")
        results.append(f"[来源 {i}] {source} 第{page}页:\n{doc.page_content}")
    return "\n\n---\n\n".join(results)

@tool
def general_knowledge(question: str) -> str:
    """当问题不需要查知识库、而是常识性问题时使用此工具"""
    # 直接返回，让 agent 用自身知识回答
    return f"这是一个一般知识问题：{question}"

tools = [search_knowledge_base, general_knowledge]
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0).bind_tools(tools)

SYSTEM = """你是一个基于知识库的问答助手。
- 涉及具体文档内容时，必须先调用 search_knowledge_base 工具
- 回答时注明信息来源（文档名和页码）
- 如果知识库中没有相关内容，如实告知用户
- 不要凭空捏造文档中的信息"""

class State(TypedDict):
    messages: Annotated[list, operator.add]

def agent_node(state):
    msgs = [SystemMessage(content=SYSTEM)] + state["messages"]
    return {"messages": [llm.invoke(msgs)]}

def router(state):
    last = state["messages"][-1]
    if hasattr(last, "tool_calls") and last.tool_calls:
        return "tools"
    return END

graph = StateGraph(State)
graph.add_node("agent", agent_node)
graph.add_node("tools", ToolNode(tools))
graph.set_entry_point("agent")
graph.add_conditional_edges("agent", router)
graph.add_edge("tools", "agent")
app = graph.compile()

def qa(question: str):
    result = app.invoke({"messages": [HumanMessage(content=question)]})
    last = result["messages"][-1]
    print(f"Q: {question}\nA: {last.content}\n")

# 测试（根据你入库的文档提问）
qa("文档中提到的主要结论是什么？")
```

### 项目 3：浏览器 Agent

**文件：`03_browser_agent/agent.py`**

```python
import asyncio
from playwright.async_api import async_playwright
from openai import AsyncOpenAI

client = AsyncOpenAI()

# 浏览器工具函数
async def navigate(page, url: str) -> str:
    await page.goto(url)
    return f"已导航至: {page.url}"

async def get_text(page) -> str:
    """获取页面主要文本内容"""
    text = await page.evaluate("""() => {
        const body = document.body.cloneNode(true);
        ['script','style','nav','footer','header'].forEach(tag => {
            body.querySelectorAll(tag).forEach(el => el.remove());
        });
        return body.innerText.slice(0, 3000);
    }""")
    return text

async def click(page, selector: str) -> str:
    try:
        await page.click(selector, timeout=5000)
        await page.wait_for_load_state("networkidle", timeout=10000)
        return f"已点击: {selector}"
    except Exception as e:
        return f"点击失败: {e}"

async def fill_input(page, selector: str, text: str) -> str:
    try:
        await page.fill(selector, text)
        return f"已填写: {selector}"
    except Exception as e:
        return f"填写失败: {e}"

async def screenshot(page, path: str = "screenshot.png") -> str:
    await page.screenshot(path=path)
    return f"截图已保存: {path}"

# 工具 schema for function calling
TOOLS = [
    {"type": "function", "function": {
        "name": "navigate",
        "description": "打开指定 URL",
        "parameters": {"type": "object", "properties": {"url": {"type": "string"}}, "required": ["url"]},
    }},
    {"type": "function", "function": {
        "name": "get_text",
        "description": "获取当前页面的文字内容",
        "parameters": {"type": "object", "properties": {}},
    }},
    {"type": "function", "function": {
        "name": "click",
        "description": "点击页面元素，使用 CSS 选择器",
        "parameters": {"type": "object", "properties": {"selector": {"type": "string"}}, "required": ["selector"]},
    }},
    {"type": "function", "function": {
        "name": "fill_input",
        "description": "在输入框中填写文字",
        "parameters": {"type": "object", "properties": {
            "selector": {"type": "string"}, "text": {"type": "string"},
        }, "required": ["selector", "text"]},
    }},
    {"type": "function", "function": {
        "name": "screenshot",
        "description": "截取当前页面截图",
        "parameters": {"type": "object", "properties": {"path": {"type": "string"}}},
    }},
]

async def browser_agent(task: str):
    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=False)  # headless=True 无界面运行
        page = await browser.new_page()

        messages = [
            {"role": "system", "content": "你是一个浏览器自动化 agent。使用提供的工具完成用户的网络任务。每次只执行一个步骤，观察结果后再决定下一步。"},
            {"role": "user", "content": task},
        ]

        import json
        for step in range(10):
            response = await client.chat.completions.create(
                model="gpt-4o",
                messages=messages,
                tools=TOOLS,
                tool_choice="auto",
            )
            msg = response.choices[0].message

            if not msg.tool_calls:
                print(f"\n✅ 任务完成:\n{msg.content}")
                break

            messages.append({"role": "assistant", "content": msg.content or "", "tool_calls": [
                {"id": tc.id, "type": "function", "function": {"name": tc.function.name, "arguments": tc.function.arguments}}
                for tc in msg.tool_calls
            ]})

            for tc in msg.tool_calls:
                func_name = tc.function.name
                args = json.loads(tc.function.arguments)
                print(f"[执行] {func_name}({args})")

                if func_name == "navigate":
                    result = await navigate(page, args["url"])
                elif func_name == "get_text":
                    result = await get_text(page)
                elif func_name == "click":
                    result = await click(page, args["selector"])
                elif func_name == "fill_input":
                    result = await fill_input(page, args["selector"], args["text"])
                elif func_name == "screenshot":
                    result = await screenshot(page, args.get("path", "screenshot.png"))
                else:
                    result = f"未知工具: {func_name}"

                print(f"[结果] {result[:200]}")
                messages.append({"role": "tool", "tool_call_id": tc.id, "content": result})

        await browser.close()

asyncio.run(browser_agent("访问 https://news.ycombinator.com 并告诉我当前排名前5的新闻标题"))
```

### 项目 4：多 Agent 研究助手

**文件：`04_research_assistant/crew.py`**

```python
from crewai import Agent, Task, Crew, Process
from crewai_tools import TavilySearchResults
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o-mini", temperature=0.3)
search_tool = TavilySearchResults(max_results=5)

# 定义专业化 agent 团队
searcher = Agent(
    role="信息检索专家",
    goal="从互联网高效检索准确、权威的信息",
    backstory="你是一名信息检索专家，擅长制定精准搜索策略，识别可信信息源。",
    tools=[search_tool],
    llm=llm,
    verbose=True,
    max_iter=5,
)

analyst = Agent(
    role="数据分析师",
    goal="对收集到的信息进行批判性分析，识别模式和关键洞察",
    backstory="你是一名资深分析师，能从海量数据中提炼核心结论，善于发现数据背后的故事。",
    llm=llm,
    verbose=True,
)

writer = Agent(
    role="报告撰写人",
    goal="将分析结论整理为结构清晰、可读性强的研究报告",
    backstory="你是一名专业作家，擅长将复杂的分析转化为清晰的书面报告。",
    llm=llm,
    verbose=True,
)

def create_research_crew(topic: str) -> Crew:
    search_task = Task(
        description=f"""对主题「{topic}」进行全面的信息检索：
        1. 搜索最新的相关数据和趋势
        2. 找到至少 5 个可信信息来源
        3. 记录每个来源的 URL 和关键数据点""",
        expected_output="包含来源 URL、关键数据和引用的结构化信息汇总",
        agent=searcher,
    )

    analysis_task = Task(
        description=f"""分析关于「{topic}」的收集信息：
        4. 识别 3-5 个核心趋势或模式
        5. 对比不同来源的观点
        6. 指出数据的局限性或不确定性""",
        expected_output="3-5 个核心发现，每个附有证据和置信度评估",
        agent=analyst,
        context=[search_task],
    )

    write_task = Task(
        description=f"""基于分析结果，撰写关于「{topic}」的专业研究报告：
        结构要求：
        - 执行摘要（150字）
        - 主要发现（3-5条，每条含数据支撑）
        - 趋势分析
        - 结论与建议
        - 参考来源""",
        expected_output="格式规范、数据充分的 Markdown 格式研究报告",
        agent=writer,
        context=[search_task, analysis_task],
        output_file="research_report.md",  # 自动保存到文件
    )

    return Crew(
        agents=[searcher, analyst, writer],
        tasks=[search_task, analysis_task, write_task],
        process=Process.sequential,
        verbose=True,
    )

if __name__ == "__main__":
    topic = "2025 年 AI Agent 在企业软件中的应用现状与趋势"
    crew = create_research_crew(topic)
    result = crew.kickoff()
    print("\n" + "="*60)
    print("研究报告已生成：research_report.md")
    print("="*60)
    print(result)
```
