# 阶段一：Agent 基础概念实践指南

## 一、基础概念

### 1.1 LLM 基础

大语言模型（LLM）通过 next-token prediction 生成文本。Agent 开发中需掌握：

- **Prompt Engineering**：System prompt 定义角色，User prompt 输入任务，Assistant prompt 可用于 few-shot 示例
- **Temperature**：控制随机性，agent 推理场景推荐 0–0.3，创意场景 0.7–1.0
- **Context window**：模型记忆上限，超出则截断——这是 memory 设计的根本动机
- **Function Calling（Tool Use）**：模型输出结构化的 JSON 表示"我要调用某工具"，由外部代码执行后结果再传回模型

### 1.2 Agent 架构

Agent = LLM + Tools + Memory + Planning 的循环系统：

```
用户输入 → [思考] → [选择工具] → [执行工具] → [观察结果] → [继续思考或输出]
```

核心范式：

- **ReAct**（Reasoning + Acting）：交替输出 Thought / Action / Observation，最常用
- **CoT**（Chain of Thought）：强制模型逐步推理再给答案，适合复杂推理
- **Reflection**：agent 对自己的输出进行批评和修正，提高准确性

### 1.3 Memory 类型

|类型|存储位置|生命周期|适用场景|
|---|---|---|---|
|短期记忆|Context window|单次对话|多轮对话上下文|
|长期记忆|向量数据库 / KV 存储|持久化|用户偏好、历史任务|
|Episodic|数据库记录|按需检索|过去的完整对话片段|
|Semantic|向量嵌入|持久化|知识库、文档|

### 1.4 Planning 策略

- **任务分解（Decomposition）**：将复杂任务拆成子任务序列
- **Tree of Thought（ToT）**：多路径探索，保留最优分支
- **Plan-and-Execute**：先生成完整计划，再逐步执行，可重新规划
- **Subgoal 设定**：设定中间检查点，避免长链推理偏移

### 步骤 2：实现 Function Calling

**文件：`02_function_calling.py`**

```python
import json
from openai import OpenAI

client = OpenAI()

# 1. 定义工具 schema
tools = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "获取指定城市的天气",
            "parameters": {
                "type": "object",
                "properties": {
                    "city": {"type": "string", "description": "城市名，如：北京"},
                    "unit": {"type": "string", "enum": ["celsius", "fahrenheit"]},
                },
                "required": ["city"],
            },
        },
    }
]

# 2. 模拟工具实现
def get_weather(city: str, unit: str = "celsius") -> dict:
    # 实际项目接真实天气 API
    return {"city": city, "temperature": 22, "condition": "晴天", "unit": unit}

# 3. 运行 tool-use 循环
def run_with_tools(user_message: str):
    messages = [{"role": "user", "content": user_message}]

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=messages,
        tools=tools,
        tool_choice="auto",
    )
    msg = response.choices[0].message

    # 4. 检查模型是否调用了工具
    if msg.tool_calls:
        messages.append(msg)  # 追加模型的 tool_call 消息

        for tool_call in msg.tool_calls:
            func_name = tool_call.function.name
            args = json.loads(tool_call.function.arguments)
            print(f"[调用工具] {func_name}({args})")

            # 5. 执行工具
            result = get_weather(**args)

            # 6. 将工具结果追加到消息
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": json.dumps(result, ensure_ascii=False),
            })

        # 7. 再次调用模型生成最终回答
        final = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
        )
        return final.choices[0].message.content
    else:
        return msg.content

print(run_with_tools("北京今天天气怎么样？"))
```

**练习**：新增一个 `calculate(expression: str)` 工具，让模型能做数学计算。

### 步骤 3：实现 ReAct 循环

**文件：`03_react_loop.py`**

```python
import re
from openai import OpenAI

client = OpenAI()

# 工具集合
def search_web(query: str) -> str:
    # 模拟搜索（实际替换为真实 API）
    return f"关于 '{query}' 的搜索结果：这是一个示例返回内容。"

def calculator(expression: str) -> str:
    try:
        return str(eval(expression))
    except Exception as e:
        return f"计算错误: {e}"

TOOLS = {"search": search_web, "calculator": calculator}

SYSTEM_PROMPT = """你是一个 ReAct agent，按以下格式思考和行动：

Thought: 分析当前情况，决定下一步
Action: search[查询内容] 或 calculator[数学表达式]
Observation: （工具返回结果，由系统填写）
... 重复以上步骤 ...
Final Answer: 最终回答

只输出 Thought 和 Action，不要伪造 Observation。"""

def parse_action(text: str):
    """从模型输出中解析 Action"""
    match = re.search(r"Action:\s*(\w+)\[(.+?)\]", text)
    if match:
        return match.group(1), match.group(2)
    return None, None

def react_agent(question: str, max_steps: int = 5):
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user",   "content": question},
    ]

    for step in range(max_steps):
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            stop=["Observation:"],  # 在 Observation 前停止，让我们填写
            temperature=0.1,
        )
        output = response.choices[0].message.content
        print(f"\n[Step {step+1}]\n{output}")

        # 检查是否已有最终答案
        if "Final Answer:" in output:
            final = output.split("Final Answer:")[-1].strip()
            print(f"\n✅ 最终答案: {final}")
            return final

        # 解析并执行 Action
        tool_name, tool_input = parse_action(output)
        if tool_name and tool_name in TOOLS:
            observation = TOOLS[tool_name](tool_input)
            print(f"Observation: {observation}")

            # 将结果追加到对话
            messages.append({"role": "assistant", "content": output})
            messages.append({"role": "user", "content": f"Observation: {observation}"})
        else:
            print("未检测到有效 Action，退出")
            break

    return "达到最大步数限制"

react_agent("2024 年诺贝尔物理学奖得主是谁？他们的主要贡献是什么？")
```

### 步骤 4：实现对话记忆

**文件：`04_memory_demo.py`**

```python
from openai import OpenAI
from collections import deque

client = OpenAI()

class ConversationAgent:
    def __init__(self, max_history: int = 10):
        self.system = "你是一个有记忆的助手，记住用户说的重要信息。"
        # 用 deque 限制历史长度，防止超出 context window
        self.history = deque(maxlen=max_history)

    def chat(self, user_input: str) -> str:
        self.history.append({"role": "user", "content": user_input})

        messages = [{"role": "system", "content": self.system}] + list(self.history)

        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=messages,
            temperature=0.3,
        )
        reply = response.choices[0].message.content
        self.history.append({"role": "assistant", "content": reply})
        return reply

    def summarize_and_compress(self):
        """当历史过长时，压缩为摘要（节省 token）"""
        if len(self.history) < 6:
            return
        history_text = "\n".join(
            f"{m['role']}: {m['content']}" for m in list(self.history)[:6]
        )
        summary_response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": "将以下对话压缩为 2-3 句摘要，保留关键信息。"},
                {"role": "user",   "content": history_text},
            ],
        )
        summary = summary_response.choices[0].message.content
        # 用摘要替换早期历史
        new_history = deque([
            {"role": "system", "content": f"[历史摘要] {summary}"},
        ] + list(self.history)[6:], maxlen=self.history.maxlen)
        self.history = new_history
        print("[记忆已压缩]")

# 使用示例
agent = ConversationAgent()
print(agent.chat("我叫张伟，我是一名 Python 后端工程师"))
print(agent.chat("我目前在学习 AI agent 开发"))
print(agent.chat("你还记得我的名字和职业吗？"))
```

**练习**：实现基于 JSON 文件的持久化记忆，让 agent 重启后仍能记住用户信息。
